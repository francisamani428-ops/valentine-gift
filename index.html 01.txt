<script>
const chat = document.getElementById("chat");

function addBubble(text, cls) {
    const div = document.createElement("div");
    div.className = "bubble " + cls;
    div.innerHTML = text;
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
}

function addOption(text, fn) {
    const div = document.createElement("div");
    div.className = "option";
    div.innerText = text;
    div.onclick = fn;
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
}

function clearOptions() {
    document.querySelectorAll(".option").forEach(o => o.remove());
}

function yesCommon() {
    clearOptions();
    addBubble("Then tell me… what is it?", "him");

    ["Cool composure","Ambiversy","Autonomous","Controversial","Simple"].forEach(item => {
        addOption(item, showBouquet);
    });
}

function noCommon() {
    clearOptions();
    addBubble("🚪", "him");
    addBubble(
        "I’m sorry… but you can’t open this door and leave my heart.<br>Maybe we could try being best friends?",
        "him"
    );
    addOption("Okay", showBouquet);
}

function showBouquet() {
    clearOptions();
    addBubble("💐", "him");
    addBubble("Like Tang San waiting through lifetimes…", "him");

    const img = document.createElement("img");
    img.src = "http://happyvalentineday2016.org/wp-content/uploads/2016/02/Valentines-Day-Hearts-Pictures.jpg";
    chat.appendChild(img);

    addBubble("Will you be my Valentine, Monicah? ❤️", "him");

    addOption("Yes 💖", yesAnswer);
    addOption("No 🙈", noAnswer);
}

function yesAnswer() {
    clearOptions();
    addBubble("💖 Then my heart has found its Xiao Wu.", "him");
    addBubble("No matter the world… I choose you.", "him");
}

function noAnswer() {
    clearOptions();
    addBubble("Even so… my heart remains.", "him");
    addBubble("Some bonds don’t fade.", "him");
}
</script>
